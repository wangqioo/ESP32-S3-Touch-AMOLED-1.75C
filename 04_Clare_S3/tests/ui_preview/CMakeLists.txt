cmake_minimum_required(VERSION 3.16)
project(clare_ui_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(APP "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CJSON_DIR "$ENV{IDF_PATH}/components/json/cJSON" CACHE PATH "ESP-IDF cJSON source directory")
if(NOT EXISTS "${CJSON_DIR}/cJSON.c")
    message(FATAL_ERROR "Set IDF_PATH or -DCJSON_DIR to the ESP-IDF cJSON directory")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${APP}/managed_components/lvgl__lvgl" lvgl)
add_executable(preview preview.cpp "${APP}/main/clare_ui.cpp" "${APP}/main/clare_mascot_0.c" "${APP}/main/clare_mascot_1.c" "${APP}/main/clare_mascot_2.c" "${APP}/main/clare_mascot_3.c" "${APP}/main/clare_mascot_4.c" "${APP}/main/clare_mascot_5.c" "${APP}/main/lv_font_clare_cjk_16.c" "${APP}/main/lv_font_clare_cjk_20.c" "${APP}/main/lv_font_clare_cjk_24.c" "${CJSON_DIR}/cJSON.c")
target_include_directories(preview PRIVATE . "${APP}/main" "${CJSON_DIR}")
target_compile_definitions(preview PRIVATE UI_FIXTURE_PATH="${CMAKE_CURRENT_LIST_DIR}/fixture.json")
target_link_libraries(preview PRIVATE lvgl)
option(UI_SANITIZE "Run native UI checks with ASan and UBSan" ON)
if(UI_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(preview PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(lvgl PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(preview PRIVATE -fsanitize=address,undefined)
endif()
